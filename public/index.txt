    function openDB() {
      return new Promise((resolve, reject) => {
        const request = indexedDB.open(DB_NAME, 2);

        request.onupgradeneeded = () => {
          if (!request.result.objectStoreNames.contains("data")) {
            request.result.createObjectStore("data");
          }
        };

        request.onsuccess = () => resolve(request.result);
        request.onerror = () => reject(request.error);
      });
    }

    function dbGet(key) {
      return new Promise((resolve, reject) => {
        const request = db
          .transaction("data", "readonly")
          .objectStore("data")
          .get(key);

        request.onsuccess = () => resolve(request.result);
        request.onerror = () => reject(request.error);
      });
    }

    function dbPut(key, value) {
      return new Promise((resolve, reject) => {
        const tx = db.transaction("data", "readwrite");
        tx.objectStore("data").put(value, key);
        tx.oncomplete = resolve;
        tx.onerror = () => reject(tx.error);
      });
    }

    function dbRemove(key) {
      return new Promise((resolve, reject) => {
        const tx = db.transaction("data", "readwrite");
        tx.objectStore("data").delete(key);
        tx.oncomplete = resolve;
        tx.onerror = () => reject(tx.error);
      });
    }

    let saveQueue = Promise.resolve();

    function saveChats() {
      if (!db) return Promise.resolve();

      const snapshot = structuredClone(chats);

      saveQueue = saveQueue
        .catch(() => {})
        .then(() => dbPut("chats", snapshot))
        .catch(() => notify(t("saveError")));

      return saveQueue;
    }

    function scheduleDraftSave() {
      clearTimeout(draftSaveTimer);
      draftSaveTimer = setTimeout(saveChats, 250);
    }

    function activeChat() {
      return chats.find(chat => chat.id === activeId);
    }

    function normalizeChat(chat) {
      chat.messages = Array.isArray(chat.messages)
        ? chat.messages
        : [];

      chat.draft = typeof chat.draft === "string"
        ? chat.draft
        : "";

      chat.draftAttachments =
        Array.isArray(chat.draftAttachments)
          ? chat.draftAttachments
          : [];

      return chat;
    }

    function makeEmptyChat() {
      const chat = {
        id: crypto.randomUUID(),
        title: t("newChat"),
        messages: [],
        draft: "",
        draftAttachments: []
      };

      chats.unshift(chat);
      activeId = chat.id;
      persistActiveId();
      saveChats();

      return chat;
    }

    /*
      Пока существует хотя бы один диалог без отправленных сообщений,
      другой пустой диалог не создаётся. Черновик этому не мешает.
    */
    function existingEmptyChat() {
      return chats.find(chat => chat.messages.length === 0);
    }

    function ensureChat() {
      return activeChat() || existingEmptyChat() || makeEmptyChat();
    }

    function captureDraft() {
      const chat = activeChat();
      if (chat) chat.draft = $("prompt").value;

      $("prompt").value = activeChat()?.draft || "";
      renderAttachments();

      const box = $("conversation");
      box.classList.remove("switching");
      void box.offsetWidth;
      box.classList.add("switching");

    function switchChat(id) {
      if (id === activeId) return;

      captureDraft();
      clearTimeout(draftSaveTimer);
      saveChats();

      activeId = id;
      persistActiveId();

      restoreDraft();
      render();
      animateConversation();

      if (innerWidth <= 700) {
        $("app").classList.remove("sidebar-open");
      }
    }

    function newChat() {
      const empty = existingEmptyChat();

      if (empty) {
        switchChat(empty.id);
        $("prompt").focus();
        return;
      }

      captureDraft();
      clearTimeout(draftSaveTimer);
      saveChats();

      makeEmptyChat();
      restoreDraft();
      render();
      animateConversation();
      $("prompt").focus();
    }

    function renderSidebar() {
      const box = $("chatList");
      box.replaceChildren();

      const query =
        $("chatSearch").value.trim().toLocaleLowerCase();

      const visible = chats.filter(chat =>
        chat.title.toLocaleLowerCase().includes(query)
      );

      if (!visible.length) {
        const empty = document.createElement("div");
        empty.className = "empty-list";
        empty.textContent =
          query ? t("noResults") : t("noChats");
        box.append(empty);
      }

      for (const chat of visible) {
        const row = document.createElement("div");
        row.className =
          "chat-row" + (chat.id === activeId ? " active" : "");

        const open = document.createElement("button");
        open.className = "chat-name";
        open.textContent = chat.title;
        open.title = chat.title;
        open.onclick = () => switchChat(chat.id);

        const remove = document.createElement("button");
        remove.className = "chat-delete";
        remove.textContent = "×";
        remove.title = t("delete");

        remove.onclick = () => {
          if (!confirm(t("delete"))) return;

          if (currentRun?.chatId === chat.id) {
            stopRun();
          }

          captureDraft();
          chats = chats.filter(item => item.id !== chat.id);

          if (activeId === chat.id) {
            activeId = chats[0]?.id || null;
            persistActiveId();
          }

          if (!activeId) {
            makeEmptyChat();
          }

          saveChats();
          restoreDraft();
          render();
          animateConversation();
        };

        row.append(open, remove);
        box.append(row);
      }
    }

        empty.textContent = t("noModels");
        items.append(empty);

      $("showMore").hidden = models.length <= menuLimit;

      for (const model of visible) {

        row.className = "model-item";

        icon.className = "model-icon";
        icon.textContent =

        const info = document.createElement("span");
        info.className = "model-info";

        const title = document.createElement("span");
        title.className = "model-title";

          settings.model = model.id;
          persistSettings();

          closeMenu();

      }

      if (menuTab === "local") {
        for (const model of visible) {
          checkCache(model.id);
        }
      }
    }

    async function checkCache(id) {
      if (
        !webllm?.hasModelInCache ||
        downloaded.has(id) ||
        checkingCache.has(id)
      ) {
        return;
      }

      checkingCache.add(id);

      try {
        downloaded.set(
          id,
          !!(await webllm.hasModelInCache(
            id,
            webllm.prebuiltAppConfig
          ))
        );
      } catch {
        // Не показываем отметку без подтверждения кэша.
      } finally {
        checkingCache.delete(id);

        if (
          !$("modelMenu").hidden &&
          menuTab === "local"
        ) {
          renderMenu();
        }
      }
    }

    async function loadCatalog(provider) {
      if (catalogs.has(provider)) return;

      if (catalogRequests.has(provider)) {
        return catalogRequests.get(provider);
      }

      catalogErrors.delete(provider);

      const task = (async () => {
        if (provider === "local") {
          webllm ||= await import(WEBLLM_URL);

          const entries =
            webllm.prebuiltAppConfig?.model_list || [];

          const models = entries
            .filter(item => item.model_id)
            .map(item => ({
              id: item.model_id,
              name: item.model_id.replace(
                /-q4f16_1-MLC$/i,
                ""
              ),
              image: false
            }));

          if (!models.length) {
            throw new Error("Каталог WebLLM пуст.");
          }

          catalogs.set(provider, models);

          if (
            settings.provider === "local" &&
            !models.some(item =>
              item.id === settings.model
            )
          ) {
            const preferred = models.find(item =>
              /Qwen2\.5-0\.5B-Instruct-q4f16_1/i
                .test(item.id)
            );

            settings.model =
              (preferred || models[0]).id;

            persistSettings();
          }

          return;
        }

        const key = connections[provider]?.key;

        if (!key) {
          throw new Error(t("keyNeeded"));
        }

        const response = await fetch(
          `/api/catalog?provider=${encodeURIComponent(provider)}`,
          {
            headers: {
              "x-provider-key": key
            }
          }
        );

        const data = await response.json();

        if (!response.ok) {
          throw new Error(
            data.error || `HTTP ${response.status}`
          );
        }

        if (!Array.isArray(data.models)) {
          throw new Error(
            "Сервер вернул некорректный каталог."
          );
        }

        catalogs.set(provider, data.models);
      })();

      catalogRequests.set(provider, task);

      if (
        !$("modelMenu").hidden &&
        menuTab === provider
      ) {
        renderMenu();
      }

      try {
        await task;
      } catch (error) {
        catalogErrors.set(provider, error.message);
        throw error;
      } finally {
        catalogRequests.delete(provider);
        renderSelectedModel();

        if (
          !$("modelMenu").hidden &&
          menuTab === provider
        ) {
          renderMenu();
        }
      }
    }

    function readDataURL(file) {
      return new Promise((resolve, reject) => {
        const reader = new FileReader();
        reader.onload = () => resolve(reader.result);
        reader.onerror = () =>
          reject(new Error("FileReader"));
        reader.readAsDataURL(file);
      });
    }

    async function extractPDF(file) {
      const pdfjs = await import(
        "https://cdn.jsdelivr.net/npm/pdfjs-dist@4.10.38/build/pdf.min.mjs"
      );

      pdfjs.GlobalWorkerOptions.workerSrc =
        "https://cdn.jsdelivr.net/npm/pdfjs-dist@4.10.38/build/pdf.worker.min.mjs";

      const pdf = await pdfjs.getDocument({
        data: await file.arrayBuffer()
      }).promise;

      const parts = [];

      for (let i = 1; i <= pdf.numPages; i++) {
        const page = await pdf.getPage(i);
        const content = await page.getTextContent();

        parts.push(
          content.items.map(item => item.str).join(" ")
        );

        if (parts.join("\n").length > 60_000) {
          break;
        }
      }

      return parts.join("\n\n").slice(0, 60_000);
    }

    async function attachFiles(files) {
      const chat = ensureChat();

      for (const file of files) {
        try {
          if (chat.draftAttachments.length >= 5) {
            throw new Error(t("fileLimit"));
          }

          if (
            ["image/png", "image/jpeg", "image/webp"]
              .includes(file.type)
          ) {
            if (file.size > 3 * 1024 * 1024) {
              throw new Error(t("fileSize"));
            }

            chat.draftAttachments.push({
              name: file.name,
              kind: "image",
              value: await readDataURL(file)
            });
          } else if (
            file.type === "application/pdf" ||
            file.name.toLowerCase().endsWith(".pdf")
          ) {
            if (file.size > 8 * 1024 * 1024) {
              throw new Error(t("fileSize"));
            }

            const value = await extractPDF(file);

            if (!value.trim()) {
              throw new Error(t("pdfEmpty"));
            }

            chat.draftAttachments.push({
              name: file.name,
              kind: "text",
              value
            });
          } else if (
            file.type.startsWith("text/") ||
            /\.(txt|md|csv|json|js|ts|py|html|css)$/i
              .test(file.name)
          ) {
            if (file.size > 500_000) {
              throw new Error(t("fileSize"));
            }

            chat.draftAttachments.push({
              name: file.name,
              kind: "text",
              value: (await file.text()).slice(0, 60_000)
            });
          } else {
            throw new Error(t("fileType"));
          }

          await saveChats();
        } catch (error) {
          notify(`${file.name}: ${error.message}`);
        }
      }

      if (activeId === chat.id) {
        renderAttachments();
      }
    }

    function messageContent(message, local) {
      let content = message.text || "";

      for (const file of message.attachments || []) {
        if (file.kind === "text") {
          content +=
            `\n\n[File: ${file.name}]\n` +
            `${file.value}\n[End of file]`;
        }
      }

      const images = (message.attachments || [])
        .filter(file => file.kind === "image");

      if (local || !images.length) {
        return content || "Analyze the attached file.";
      }

      return [
        {
          type: "text",
          text: content || "Analyze this image."
        },
        ...images.map(file => ({
          type: "image_url",
          image_url: { url: file.value }
        }))
      ];
    }

    async function getLocalEngine(model, run) {
      if (!navigator.gpu) {
        throw new Error(t("webgpu"));
      }

      if (engine && engineModel === model) {
        return engine;
      }

      if (!enginePromise) {
        enginePromise = (async () => {
          if (engine) {
            await engine.unload();
            engine = null;
            engineModel = "";
          }

          webllm ||= await import(WEBLLM_URL);

          const created =
            await webllm.CreateMLCEngine(model, {
              initProgressCallback: progress => {
                if (run.cancelled) return;

                const percent =
                  Number.isFinite(progress.progress)
                    ? ` ${Math.round(progress.progress * 100)}%`
                    : "";

                run.show(
                  `${t("preparing")}${percent}\n` +
                  `${progress.text || ""}`
                );
              }
            });

          engine = created;
          engineModel = model;
          downloaded.set(model, true);

          return created;
        })();
      }

      try {
        return await enginePromise;
      } finally {
        enginePromise = null;
      }
    }

    function readCompatibleAnswer(data) {
      const content = data?.choices?.[0]?.message?.content;

      if (typeof content === "string") {
        return content;
      }

      if (Array.isArray(content)) {
        return content
          .map(part => part.text || "")
          .join("");
      }

      return "";
    }

    async function answer() {
      if (currentRun) {
        notify(t("busy"));
        return;
      }

      const chat = activeChat();

      if (
        !chat ||
        chat.messages.at(-1)?.role !== "user"
      ) {
        return;
      }

      const provider = settings.provider;
      const model = settings.model;

      if (!model) {
        notify(t("chooseModel"));
        return;
      }

      const connection =
        provider === "local"
          ? null
          : structuredClone(connections[provider]);

      if (
        provider !== "local" &&
        provider !== "custom" &&
        !connection?.key
      ) {
        notify(t("keyNeeded"));
        openSettings(provider);
        return;
      }

      if (
        provider === "custom" &&
        (
          !connection?.model ||
          !validEndpoint(connection.endpoint)
        )
      ) {
        notify(t("customNotConfigured"));
        openSettings("custom");
        return;
      }

      const run = {
        cancelled: false,
        controller: new AbortController(),
        chatId: chat.id,
        show: () => {}
      };

      currentRun = run;
      $("send").textContent = "■";
      $("send").classList.add("stop");

      const reply = {
        id: crypto.randomUUID(),
        role: "assistant",
        provider,
        model,
        text: ""
      };

      chat.messages.push(reply);

      if (activeId === chat.id) {
        renderMessages();
      }

      run.show = value => {
        if (activeId !== chat.id) return;

        const bubble = [...$("conversation")
          .querySelectorAll("[data-message-id]")]
          .find(node => node.dataset.messageId === reply.id);

        if (bubble) {
          bubble.textContent = value;
          $("conversation").scrollTop =
            $("conversation").scrollHeight;
        }
      };

      try {
        const messages = chat.messages
          .slice(0, -1)
          .slice(-20)
          .map(message => ({
            role: message.role,
            content:
              message.role === "user"
                ? messageContent(
                    message,
                    provider === "local"
                  )
                : message.text
          }));

        if (provider === "local") {
          run.show(t("preparing"));

          const current =
            await getLocalEngine(model, run);

          if (!run.cancelled) {
            const stream =
              await current.chat.completions.create({
                messages,
                stream: true
              });

            run.show("…");

            for await (const packet of stream) {
              if (run.cancelled) break;

              const piece =
                packet.choices?.[0]?.delta?.content;

              if (piece) {
                reply.text += piece;
                run.show(reply.text);
              }
            }
          }
        } else {
          run.show("…");

          const custom = provider === "custom";
          const headers = {
            "Content-Type": "application/json"
          };

          if (custom) {
            if (connection.key) {
              headers.Authorization =
                `Bearer ${connection.key}`;
            }
          } else {
            headers["x-provider-key"] =
              connection.key;
          }

          const response = await fetch(
            custom
              ? connection.endpoint
              : "/api/chat",
            {
              method: "POST",
              signal: run.controller.signal,
              headers,
              body: JSON.stringify(
                custom
                  ? {
                      model,
                      messages,
                      stream: false
                    }
                  : {
                      provider,
                      model,
                      messages
                    }
              )
            }
          );

          let data;

          try {
            data = await response.json();
          } catch {
            throw new Error(
              `HTTP ${response.status}: invalid JSON response`
            );
          }

          if (!response.ok) {
            const detail =
              data?.error?.message ||
              data?.error ||
              `HTTP ${response.status}`;

            throw new Error(
              typeof detail === "string"
                ? detail
                : JSON.stringify(detail)
            );
          }

          if (!run.cancelled) {
            reply.text = custom
              ? readCompatibleAnswer(data)
              : data.text || "";
          }
        }

        if (
          !run.cancelled &&
          !reply.text.trim()
        ) {
          throw new Error(t("emptyAnswer"));
        }
      } catch (error) {
        if (!run.cancelled) {
          notify(`${t("failed")}: ${error.message}`);
        }
      } finally {
        if (run.cancelled && reply.text) {
          reply.text += `\n\n${t("stopped")}`;
        }

        if (!reply.text) {
          const index = chat.messages.indexOf(reply);

          if (index !== -1) {
            chat.messages.splice(index, 1);
          }
        }

        if (currentRun === run) {
          currentRun = null;
        }

        $("send").textContent = "↑";
        $("send").classList.remove("stop");

        await saveChats();

        if (activeId === chat.id) {
          renderMessages();
        }

        if (!$("modelMenu").hidden) {
          renderMenu();
        }
      }
    }

    async function sendMessage() {
      if (currentRun) {
        stopRun();
        return;
      }

      const chat = ensureChat();
      captureDraft();

      const value = chat.draft.trim();
      const attachments = chat.draftAttachments;

      if (!value && !attachments.length) return;

      const hasImage = attachments.some(file =>
        file.kind === "image"
      );

      if (
        hasImage &&
        settings.provider === "local"
      ) {
        notify(t("imageLocal"));
        return;
      }

      if (hasImage) {
        const chosen = providerModels(
          settings.provider
        ).find(model =>
          model.id === settings.model
        );

        if (!chosen?.image) {
          notify(t("imageUnsupported"));
          return;
        }
      }

      if (!settings.model) {
        notify(t("chooseModel"));
        return;
      }

      if (
        settings.provider !== "local" &&
        settings.provider !== "custom" &&
        !connections[settings.provider]?.key
      ) {
        notify(t("keyNeeded"));
        openSettings(settings.provider);
        return;
      }

      if (
        settings.provider === "custom" &&
        !validEndpoint(connections.custom.endpoint)
      ) {
        notify(t("customNotConfigured"));
        openSettings("custom");
        return;
      }

      const first = chat.messages.length === 0;

      chat.messages.push({
        id: crypto.randomUUID(),
        role: "user",
        text: value,
        attachments
      });

      if (first) {
        chat.title =
          (
            value ||
            attachments[0]?.name ||
            t("newChat")
          ).slice(0, 48);
      }

      chat.draft = "";
      chat.draftAttachments = [];
      $("prompt").value = "";

      clearTimeout(draftSaveTimer);
      await saveChats();

      render();
      answer();
    }

    async function preparePhoto(file) {
      if (file.size > 12 * 1024 * 1024) {
        throw new Error(t("photoSize"));
      }

      if (
        !["image/png", "image/jpeg", "image/webp"]
          .includes(file.type)
      ) {
        throw new Error(t("photoError"));
      }

      let image;
      let objectURL;

      try {
        if ("createImageBitmap" in window) {
          image = await createImageBitmap(file);
        } else {
          throw new Error("fallback");
        }
      } catch {
        objectURL = URL.createObjectURL(file);

        image = await new Promise((resolve, reject) => {
          const element = new Image();
          element.onload = () => resolve(element);
          element.onerror = () =>
            reject(new Error(t("photoError")));
          element.src = objectURL;
        });
      }

      try {
        const width = image.width || image.naturalWidth;
        const height = image.height || image.naturalHeight;

        if (!width || !height) {
          throw new Error(t("photoError"));
        }

        const scale = Math.min(
          1,
          1600 / Math.max(width, height)
        );

        const canvas = document.createElement("canvas");
        canvas.width = Math.max(1, Math.round(width * scale));
        canvas.height = Math.max(1, Math.round(height * scale));

        const context = canvas.getContext("2d");

        if (!context) {
          throw new Error(t("photoError"));
        }

        context.drawImage(
          image,
          0,
          0,
          canvas.width,
          canvas.height
        );

        return await new Promise((resolve, reject) => {
          canvas.toBlob(
            blob => blob
              ? resolve(blob)
              : reject(new Error(t("photoError"))),
            "image/jpeg",
            .82
          );
        });
      } finally {
        image.close?.();

        if (objectURL) {
          URL.revokeObjectURL(objectURL);
        }
      }
    }

    function showPhoto(blob) {
      if (photoURL) {
        URL.revokeObjectURL(photoURL);
      }

      photoURL =
        blob ? URL.createObjectURL(blob) : null;

      document.documentElement.style.setProperty(
        "--photo",
        photoURL
          ? `url("${photoURL}")`
          : "none"
      );
    }

      captureDraft();
      clearTimeout(draftSaveTimer);
      saveChats();
    });

    async function start() {
      persistSettings();
      applyTheme();
      applyLanguage();

      try {
        db = await openDB();

        const stored = await dbGet("chats");

        chats = Array.isArray(stored)
          ? stored.map(normalizeChat)
          : [];

        const storedId =
          localStorage.getItem(ACTIVE_KEY);

        activeId =
          chats.find(chat => chat.id === storedId)?.id ||
          chats[0]?.id ||
          null;

        if (!activeId) {
          makeEmptyChat();
        }

        persistActiveId();

        const savedConnections =
          await dbGet("provider-connections");

        if (
          savedConnections &&
          typeof savedConnections === "object"
        ) {
          for (const provider of [
            "huggingface",
            "gemini",
            "custom"
          ]) {
            if (
              savedConnections[provider] &&
              typeof savedConnections[provider] === "object"
            ) {
              connections[provider] = {
                ...connections[provider],
                ...savedConnections[provider]
              };
            }
          }
        } else {
          // Перенос ключей из предыдущей версии.
          const oldKeys = await dbGet("provider-keys");

          if (oldKeys) {
            connections.huggingface.key =
              typeof oldKeys.huggingface === "string"
                ? oldKeys.huggingface
                : "";

            connections.gemini.key =
              typeof oldKeys.gemini === "string"
                ? oldKeys.gemini
                : "";
          }

          let migrated = !!oldKeys;

          for (const provider of [
            "huggingface",
            "gemini"
          ]) {
            const sessionKey =
              sessionStorage.getItem(
                `extend-key-${provider}`
              );

            if (
              !connections[provider].key &&
              sessionKey
            ) {
              connections[provider].key =
                sessionKey;
              migrated = true;
            }

            sessionStorage.removeItem(
              `extend-key-${provider}`
            );
          }

          if (migrated) {
            await dbPut(
              "provider-connections",
              connections
            );
          }
        }

        const photo = await dbGet("theme-photo");

        if (photo instanceof Blob) {
          showPhoto(photo);
        }

        restoreDraft();
        render();
      } catch (error) {
        notify(`${t("saveError")} ${error.message}`);
      }

      loadCatalog("local").catch(error => {
        notify(`${t("failed")}: ${error.message}`);
      });
    }

    start();
  </script>
